// PAGINA B (Ontvanger)
const kanaal = new BroadcastChannel('mijn_kanaal');
kanaal.onmessage = (event) => {
    console.log('Ontvangen data:', event.data);
};

// PAGINA A (Verzender)

// PAGINA B (Ontvanger)
window.addEventListener('storage', (event) => {
    if (event.key === 'laatste_actie') {
        const data = JSON.parse(event.newValue);
        console.log('Verandering opgemerkt:', data);
    }
});
